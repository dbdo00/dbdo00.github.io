单张图片段落（默认 reader 会转成 Figure）

-- gallery.lua — pandoc Lua filter
--
-- 把 markdown 中"纯图片段落"的一组图片转换为相册：
--   同一段落内多张图片 -> <div class="gallery"> <figure>...</figure> ... </div>
-- 仿照 hugo-theme-stack 的相册功能：
--   - 构建期用 pandoc.image.size（pandoc >= 3.1.13）解析图片宽高，
--     按宽高比计算 flex-grow/flex-basis，实现弹性等高排布（对应 layouts/_markup/render-image.html）
--   - 段落级别的"纯图片"判断在 AST 上完成（对应 assets/ts/gallery.ts 的 no-text 检查）
--   - 图片结构用原生 pandoc AST（Figure/Link/Image）生成，转义由 writer 负责，
--     无需手写 HTML 或 escape 函数
--   - 自动向文档注入 .gallery 的 CSS（无 JavaScript，不依赖外部资源）
--
-- 用法：
--   pandoc demo.md --lua-filter=gallery.lua -s -o demo.html

os.setlocale('C')

-- ---------------------------------------------------------------------------
-- 读取图片宽高
--
-- 用 pandoc.image.size（pandoc >= 3.1.13）解析二进制内容，返回 {width, height}。
-- 失败（非图片 / 尺寸无法确定 / 资源获取失败）时返回 nil。
-- ---------------------------------------------------------------------------

local function read_dimensions(src)
  -- pandoc.mediabag.fetch: 先查 mediabag，再按 resource-path 从本地/远程获取；
  -- 资源缺失时抛错，用 pcall 捕获后把错误信息打印到 stderr
  local ok_fetch, mime, contents = pcall(pandoc.mediabag.fetch, src)
  if not ok_fetch then
    io.stderr:write('[gallery] warning: fetch "'
      .. src .. '" failed: ' .. tostring(mime) .. '\n')
    return nil
  end
  if not contents then return nil end

  local ok_size, info = pcall(pandoc.image.size, contents)
  if not ok_size then
    io.stderr:write('[gallery] warning: pandoc.image.size for "'
      .. src .. '" failed: ' .. tostring(info) .. '\n')
    return nil
  end
  if info and info.width and info.height then
    return info.width, info.height
  end
  return nil
end

-- ---------------------------------------------------------------------------
-- flex 值计算（对应 render-image.html: data-flex-grow / data-flex-basis）
--   flex-grow  = w * 100 / h
--   flex-basis = w * 240 / h  px
-- ---------------------------------------------------------------------------

local function flex_grow(w, h)
  return math.floor(w * 100 / h + 0.5)
end

local function flex_basis(w, h)
  return math.floor(w * 240 / h + 0.5) .. 'px'
end

-- ---------------------------------------------------------------------------
-- 单张图片 -> pandoc.Figure（含 Link 包裹与 figcaption）
-- ---------------------------------------------------------------------------

local function image_to_figure(img)
  local src = img.src
  local alt = pandoc.utils.stringify(img.caption)
  local title = img.title or ''

  local w, h = read_dimensions(src)
  if not w then
    io.stderr:write('[gallery] warning: cannot get dimensions of "'
      .. src .. '", flex layout degraded\n')
  end

  -- <img ...>：alt 作为 caption，title 作为 img 的 title 属性
  local img_attrs = { loading = 'lazy' }
  if w and h then
    img_attrs.width = tostring(w)
    img_attrs.height = tostring(h)
  end
  local image = pandoc.Image(pandoc.Str(alt), src, title, img_attrs)

  -- 用 Link 包裹，点击看大图（target=_blank 由 writer 输出）
  local link = pandoc.Link({ image }, src, '', { target = '_blank' })

  -- figcaption：优先 title 否则 alt；都没有则不生成
  local caption = title ~= '' and title or alt
  local cap = caption ~= '' and pandoc.Caption(pandoc.Str(caption)) or pandoc.Caption{}

  -- <figure class="gallery-image" style="flex-grow:..;flex-basis:..">
  local fig_attrs = { class = 'gallery-image' }
  if w and h then
    fig_attrs.style = string.format('flex-grow:%d;flex-basis:%s', flex_grow(w, h), flex_basis(w, h))
  end

  return pandoc.Figure({ pandoc.Plain({ link }) }, cap, fig_attrs)
end

-- ---------------------------------------------------------------------------
-- 判断段落是否为"纯图片"（对应 gallery.ts 的 no-text 检查）
-- ---------------------------------------------------------------------------

local pure_image_separators = { Space = true, SoftBreak = true, LineBreak = true }

local function is_pure_image_para(content)
  local count = 0
  for _, item in ipairs(content) do
    if item.t == 'Image' then
      count = count + 1
    elseif not pure_image_separators[item.t] then
      return false
    end
  end
  return count > 0
end

local function build_gallery(images)
  local figures = {}
  for _, img in ipairs(images) do
    figures[#figures + 1] = image_to_figure(img)
  end
  return pandoc.Div(figures, { class = 'gallery' })
end

-- ---------------------------------------------------------------------------
-- 注入的 CSS（移植自 assets/scss/partials/layout/article.scss）
-- ---------------------------------------------------------------------------

local GALLERY_CSS = [[
<style>
.gallery {
    position: relative;
    display: flex;
    flex-direction: row;
    justify-content: center;
    margin: 1.5em 0;
    gap: 8px;
}
.gallery figure {
    margin: 0;
    min-width: 0;
}
.gallery a {
    display: block;
}
.gallery img {
    max-width: 100%;
    height: auto;
    display: block;
}
.gallery figcaption {
    text-align: center;
    font-size: .8em;
    color: #666;
    margin-top: .35em;
}
</style>
]]

-- ---------------------------------------------------------------------------
-- filter 主体
-- ---------------------------------------------------------------------------

local function is_html_format()
  return FORMAT and FORMAT:match('html')
end

function Para(para)
  if not is_html_format() then return nil end
  if not is_pure_image_para(para.content) then return nil end

  local images = {}
  for _, item in ipairs(para.content) do
    if item.t == 'Image' then
      images[#images + 1] = item
    end
  end
  return { build_gallery(images) }
end

-- 兼容默认 reader：单图段落会被 implicit_figures 自动转成 Figure。
-- 把"内容只含图片"的 Figure 转成单图相册。
function Figure(figure)
  if not is_html_format() then return nil end

  local images = {}
  for _, block in ipairs(figure.content) do
    if (block.t == 'Plain' or block.t == 'Para') and is_pure_image_para(block.content) then
      for _, item in ipairs(block.content) do
        if item.t == 'Image' then images[#images + 1] = item end
      end
    end
  end

  if #images > 0 then
    return { build_gallery(images) }
  end
  return nil
end

function Pandoc(doc)
  if not is_html_format() then return doc end
  doc.blocks:insert(1, pandoc.RawBlock('html', GALLERY_CSS))
  return doc
end

return { Para = Para, Figure = Figure, Pandoc = Pandoc }
